### 1. Create the Main Project Folder
mkdir django_1
cd django_1

### 2. Set Up the Virtual Environment
python3 -m venv venv

# Activate the environment (use the command for your OS)
# On Windows:
venv\Scripts\activate.bat
# On macOS/Linux:
source venv/bin/activate

----make directory src----
# mkdir src

### 3. Install Django --- go to main dir
pip install django

# You can verify the correct Python is being used
# On Windows: where python
# On macOS/Linux: which python

### 4. Create the Source Directory  - in Django in django
# This is where your project's code will live.
mkdir src
cd src

### 5. Start the Django Project
# Use the project name and a '.' to create it in the current (src) directory.
django-admin startproject django_1 .

### 6. Run the Development Server
# Make sure you are in the 'src' directory where manage.py is located.
python manage.py runserver

# Open your browser and go to http://127.0.0.1:8000/ to see your site.
# The server will automatically refresh when you save your files.
