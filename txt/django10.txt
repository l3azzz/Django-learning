class Products(models.Model):
    
    logo = models.FileField(upload_to="products/logo")
    title = models.CharField(max_length=128)
    main_title = models.CharField(max_length=128)
    description = models.TextField()
    main_description = models.TextField()
    image = models.ImageField(upload_to="products/image")
    hero_image = models.ImageField(upload_to="products/hero_image")
    bg_color = models.CharField(max_length=128)

    class Meta:
        db_table = "web_products"
        ordering = ['id']
